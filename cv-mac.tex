  \_penalty-1000\_par\_bigskip

\_def\_incventry #1{%
  \_par\_medskip
  \_bgroup
    \_noindent\_llap{%
      \_vtop to 0pt{%
        \_hsize=\_sectionrulewidth
        \_advance\_hsize\_postsectionruleskip
        \_leftskip=0pt plus 1fil
        \_rightskip=\_postsectionruleskip
        \_parfillskip=0pt
        \_let\nl=\_break
        \_noindent\_scantextokens{#1}\_vss
      }%
    }%
    \_advance\_leftskip by \_sectionrulewidth
    \_advance\_leftskip by \_postsectionruleskip
    \_ignorespaces\_noindent
    \_cventryopentrue
}

  \_line{%
    \_vbox{\_hsize=.58\_hsize

      \_leftline{\_Secondcolor\_subtitlefont\_it\_cvtitle}
    }\_hfill
    \_vbox{\_hsize=.34\_hsize\_Secondcolor\_parindent=0pt\_typoscale[900/900]
      \_rightskip=0pt plus 1fil
      \_parfillskip=0pt
      \_noindent\_llap{\_picheight=2ex\_lower.1ex\_inspic{location-symbol.pdf}}\ \_maplink\_par
      \_noindent\_llap{\_picheight=2ex\_lower.1ex\_inspic{phone-symbol.pdf}}\ \_phonelink\_par
      \_noindent\_llap{\_picheight=1.3ex\_inspic{email-symbol.pdf}}\ \_maillink\_par
      \_noindent\_llap{\_picheight=2ex\_inspic{git-symbol.pdf}}\ \_gitlink\_par
    }\_kern1em
  }\_vskip1.5cm
}
